Question 7: Price differences between different sizes of the same pizza (SELF JOIN)
SELECT distinct p1.pizza_type_id, p1.size as size1, p2.size as size2, p1.price as price1, p2.price as price2, Round(p2.price-p1.price,2) as price_differences FROM pizzas p1 inner join pizzas p2 on p1.pizza_type_id=p2.pizza_type_id and case when p1.size='S' then 1 when p1.size='M' then 2 when p1.size='L' then 3 end < case when p2.size='S' then 1 when p2.size='M' then 2 when p2.size='L' then 3 end
